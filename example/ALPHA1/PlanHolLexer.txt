\hshead {
    module Hol.ALPHA1.Front.Analyzer.Lexer where

    import Hol.ALPHA1.Front.Analyzer.Grammar
    import Hol.ALPHA1.Front.Header
}

\target {
    token-type: "Either (Int, Int) Token"
    lexer-name: "runAladdinLexerRaw"
}

\define $small_letter = 'a'-'z'
\define $big_letter = 'A'-'Z'
\define $digit = '0'-'9'
\define $printable = ' '-'~'
\define $comment_char = '\n' $printable
\define $Escape = "\\n" + "\\\\" + "\\\"" + "\\\'" + "\\t"
\define $StringChar = [$printable \ '\\' \ '\"'] + $Escape
\define $CharChar = [$printable \ '\\' \ '\''] + $Escape
\define $SmallId = [$small_letter] [$small_letter $digit '_' $big_letter]*
\define $LargeId = [$big_letter] [$small_letter $digit '_' $big_letter]* + "_"

\xmatch ".":
    Right (T_dot (SLoc (row1, col1) (row2, col2)))
\xmatch "->":
    Right (T_arrow (SLoc (row1, col1) (row2, col2)))
\xmatch "=>":
    Right (T_fatarrow (SLoc (row1, col1) (row2, col2)))
\xmatch "(":
    Right (T_lparen (SLoc (row1, col1) (row2, col2)))
\xmatch ")":
    Right (T_rparen (SLoc (row1, col1) (row2, col2)))
\xmatch "[":
    Right (T_lbracket (SLoc (row1, col1) (row2, col2)))
\xmatch "]":
    Right (T_rbracket (SLoc (row1, col1) (row2, col2)))
\xmatch "?-":
    Right (T_quest (SLoc (row1, col1) (row2, col2)))
\xmatch ",":
    Right (T_comma (SLoc (row1, col1) (row2, col2)))
\xmatch "=>":
    Right (T_fatarrow (SLoc (row1, col1) (row2, col2)))
\xmatch ":-":
    Right (T_if (SLoc (row1, col1) (row2, col2)))
\xmatch "=":
    Right (T_Eq (SLoc (row1, col1) (row2, col2)))
\xmatch "is":
    Right (T_arithmetic_predicate (SLoc (row1, col1) (row2, col2)) AP_Is)
\xmatch "=:=":
    Right (T_arithmetic_predicate (SLoc (row1, col1) (row2, col2)) AP_Eq)
\xmatch "=\\=":
    Right (T_arithmetic_predicate (SLoc (row1, col1) (row2, col2)) AP_Ne)
\xmatch "<":
    Right (T_arithmetic_predicate (SLoc (row1, col1) (row2, col2)) AP_Lt)
\xmatch "=<":
    Right (T_arithmetic_predicate (SLoc (row1, col1) (row2, col2)) AP_Le)
\xmatch ">":
    Right (T_arithmetic_predicate (SLoc (row1, col1) (row2, col2)) AP_Gt)
\xmatch ">=":
    Right (T_arithmetic_predicate (SLoc (row1, col1) (row2, col2)) AP_Ge)
\xmatch "+":
    Right (T_plus (SLoc (row1, col1) (row2, col2)))
\xmatch "-":
    Right (T_minus (SLoc (row1, col1) (row2, col2)))
\xmatch "*":
    Right (T_star (SLoc (row1, col1) (row2, col2)))
\xmatch "/":
    Right (T_slash (SLoc (row1, col1) (row2, col2)))
\xmatch "//":
    Right (T_quotient (SLoc (row1, col1) (row2, col2)))
\xmatch "div":
    Right (T_div (SLoc (row1, col1) (row2, col2)))
\xmatch "mod":
    Right (T_mod (SLoc (row1, col1) (row2, col2)))
\xmatch "rem":
    Right (T_rem (SLoc (row1, col1) (row2, col2)))
\xmatch "s":
    Right (T_succ (SLoc (row1, col1) (row2, col2)))
\xmatch "pi":
    Right (T_pi (SLoc (row1, col1) (row2, col2)))
\xmatch "sigma":
    Right (T_sigma (SLoc (row1, col1) (row2, col2)))
\xmatch ";":
    Right (T_semicolon (SLoc (row1, col1) (row2, col2)))
\xmatch "!":
    Right (T_cut (SLoc (row1, col1) (row2, col2)))
\xmatch "true":
    Right (T_true (SLoc (row1, col1) (row2, col2)))
\xmatch "fail":
    Right (T_fail (SLoc (row1, col1) (row2, col2)))
\xmatch "debug":
    Right (T_debug (SLoc (row1, col1) (row2, col2)))
\xmatch "\\":
    Right (T_bslash (SLoc (row1, col1) (row2, col2)))
\xmatch "::":
    Right (T_cons (SLoc (row1, col1) (row2, col2)))
\xmatch "kind":
    Right (T_kind (SLoc (row1, col1) (row2, col2)))
\xmatch "type":
    Right (T_type (SLoc (row1, col1) (row2, col2)))
\xmatch $SmallId:
    Right (T_smallid (SLoc (row1, col1) (row2, col2)) this)
\xmatch $LargeId:
    Right (T_largeid (SLoc (row1, col1) (row2, col2)) this)
\xmatch [$digit]+:
    readLiteralToken (row1, col1) (T_nat_lit (SLoc (row1, col1) (row2, col2))) this
\xmatch "\"" $StringChar* "\"":
    readLiteralToken (row1, col1) (T_str_lit (SLoc (row1, col1) (row2, col2))) this
\xmatch "\'" $CharChar "\'":
    readLiteralToken (row1, col1) (T_chr_lit (SLoc (row1, col1) (row2, col2))) this
\xmatch [' ' '\n']+: skip
\xmatch "%" [$comment_char \ '\n']*: skip
\xmatch "(*" ([$comment_char \ '*'] + "*"+ [$comment_char \ '*' \ ')'])* "*"+ ")": skip

\hstail {
    runAladdinLexer :: String -> Either (Int, Int) [Token]
    runAladdinLexer src = runAladdinLexerRaw src >>= sequence

    readLiteralToken :: Read a => (Int, Int) -> (a -> Token) -> String -> Either (Int, Int) Token
    readLiteralToken start makeToken src
        = case reads src of
            [(value, "")] -> Right (makeToken value)
            _ -> Left start
}
